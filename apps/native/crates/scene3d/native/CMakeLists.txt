cmake_minimum_required(VERSION 3.16)
project(pm-scene3d-cad LANGUAGES CXX)
find_package(OpenCASCADE REQUIRED)
add_executable(pm-scene3d-cad cad.cpp)
target_compile_features(pm-scene3d-cad PRIVATE cxx_std_17)
target_include_directories(pm-scene3d-cad PRIVATE ${OpenCASCADE_INCLUDE_DIR})
target_link_directories(pm-scene3d-cad PRIVATE ${OpenCASCADE_LIBRARY_DIR})
target_link_libraries(pm-scene3d-cad PRIVATE ${OpenCASCADE_FoundationClasses_LIBRARIES} ${OpenCASCADE_ModelingData_LIBRARIES} ${OpenCASCADE_ModelingAlgorithms_LIBRARIES} ${OpenCASCADE_ApplicationFramework_LIBRARIES} ${OpenCASCADE_DataExchange_LIBRARIES})
install(TARGETS pm-scene3d-cad RUNTIME DESTINATION bin)
